function snookerRemainingPoints(d){
  // Version 8.8: Remaining Points is derived only from balls/ball opportunities
  // physically left in the frame sequence. Foul awards never reduce this value.
  if(!d||d.snookerFrameComplete)return 0;
  const reds=Math.max(0,Number(d.snookerRedsRemaining)||0);
  const colors=snookerColorData();
  if(d.snookerPhase==='reds'){
    // Each red still on the table can yield Red (1) + Black (7) = 8.
    // If a red has just been potted, one colour opportunity is currently due (+7 max).
    // The six final colours always contribute 27 while the reds phase is active.
    return Math.max(0,(reds*8)+(d.snookerTarget==='color'?7:0)+27);
  }
  const idx=Math.max(0,Math.min(colors.length,Number(d.snookerColorIndex)||0));
  return colors.slice(idx).reduce((sum,x)=>sum+Number(x[1]||0),0);
}
